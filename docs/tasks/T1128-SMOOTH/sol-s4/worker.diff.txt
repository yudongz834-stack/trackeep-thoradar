+  const tags = /<!--[\s\S]*?(?:--!?>|$)|<[!?][^>]*>|<(\/?)([a-z][\w:-]*)(?=[\t\n\f\r />])(?:[^"'<>]|"[^"]*"|'[^']*')*>/gi;
